app.post('/api/products', async (req, res) => {

    const { product_name, price, quantity, image_filename, snnumber, detail } = req.body;
    const sql = 'INSERT INTO SQLWORK (product_name, price, quantity, image_filename, snnumber, detail) VALUES (?, ?, ?, ?, ?, ?)';

app.put('/api/products/:id/buy', async (req, res) => {
  try {
    const { id } = req.params;
    const [rows] = await pool.query('SELECT quantity FROM SQLWORK WHERE id = ?', [id]);
    if (rows.length === 0) return res.status(404).json({ error: 'Product not found' });
    
    const currentQty = rows[0].quantity;
    if (currentQty <= 0) {
      return res.status(400).json({ error: 'Product is out of stock' });
    }

    await pool.query('UPDATE SQLWORK SET quantity = quantity - 1 WHERE id = ?', [id]);
    res.json({ success: true, message: 'Purchase successful' });
  } catch (e) {
    console.error("Buy Error:", e);
    res.status(500).json({ error: 'Failed to process purchase' });
  }
});
